import pygame import random pygame.init() WIDTH, HEIGHT = 480, 360 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("STEP 3 - 슈팅게임") clock = pygame.time.Clock() # 색상 BG_COLOR = (245, 245, 245) PLAYER_COLOR = (27, 31, 29) ENEMY_COLOR = (63, 102, 80) BULLET_COLOR = (86, 94, 88) # 게임 변수 player_x = WIDTH // 2 - 15 player_y = 330 player_speed = 7 bullets = [] enemies = [] score = 0 spawn_timer = 0 running = True game_over = False while running: clock.tick(60) # 60 FPS 고정 for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.KEYDOWN and not game_over: if event.key == pygame.K_SPACE: bullets.append(pygame.Rect(player_x + 13, player_y, 4, 8)) if not game_over: # 키 연속 누름 이동 keys = pygame.key.get_pressed() if keys[pygame.K_LEFT]: player_x = max(0, player_x - player_speed) if keys[pygame.K_RIGHT]: player_x = min(WIDTH - 30, player_x + player_speed) # 적 생성 (일정 프레임 간격) spawn_timer += 1 if spawn_timer >= 60: spawn_timer = 0 enemies.append(pygame.Rect(random.randint(0, WIDTH - 30), -20, 30, 20)) # 총알 이동 for bullet in bullets[:]: bullet.y -= 7 if bullet.y < 0: bullets.remove(bullet) # 적 이동 및 바닥 도착 검사 player_rect = pygame.Rect(player_x, player_y, 30, 20) for enemy in enemies[:]: enemy.y += 2 # [게임 종료 조건] 적이 바닥에 닿거나 기체와 부딪힘 if enemy.y + enemy.height >= HEIGHT or enemy.colliderect(player_rect): game_over = True # 총알 맞춤 판정 for bullet in bullets[:]: if enemy.colliderect(bullet): enemies.remove(enemy) bullets.remove(bullet) score += 10 break # 화면 그리기 screen.fill(BG_COLOR) if not game_over: pygame.draw.rect(screen, PLAYER_COLOR, (player_x, player_y, 30, 20)) for enemy in enemies: pygame.draw.rect(screen, ENEMY_COLOR, enemy) for bullet in bullets: pygame.draw.rect(screen, BULLET_COLOR, bullet) else: font = pygame.font.SysFont(None, 36) text = font.render(f"Game Over! Score: {score}", True, (200, 0, 0)) screen.blit(text, (WIDTH // 2 - 130, HEIGHT // 2 - 20)) pygame.display.flip() pygame.quit()
